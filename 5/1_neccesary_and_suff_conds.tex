\section{Conditions for Metric Representation of AD Structures}
% \addcontentsline{toc}{section}{Theorem: Necessary and sufficient conditions for additive difference metrics}
\begin{thm}[Conditions for Metric\footnotemark Representation of AD Structures]\label{thm-necc-suff-conds} 
  \footnotetext{Metric representation that are additive on the coordinate axes}
  Suppose $\langle \ps{A}, \succsim \rangle$ is an additive-difference proximity structure such that $f$ is the additive-difference representation from \cref{thm-repr-add-diff-struct}, i.e.,
	\[
    f(\ps{a}, \ps{b}) = \sum_{i=1}^{n} F_i(|\varphi_i(\ps{a}) - \varphi_i(\ps{b})|).
	\]
	Suppose further that the domain of each $F_i$ is a real interval and $F_i(0) = 0$, $i = 1, \ldots, n$ (through transformation, since $F_i$ are interval scales).

  For $\langle \ps{A}, \succsim \rangle$ to be compatible with a metric, and additionally be additive on the coordinate axes,
	\begin{enumerate}
		\item it is necessary that there exists a continuous, convex function $F$, such that $F_i(x) = F(t_i x), t_i > 0,$ for all $x$ in the domain of $F_i$, $i = 1, \ldots, n$; and \label{thm-necc-cond}
		\item sufficient that, in addition (to the conditions in 1.), the function $\log F(e^x)$ is convex. \label{thm-suff-cond}
	\end{enumerate}
\end{thm}

\begin{rem}[Additivity on the Coordinate Axes] \gls{ps-abc-between}
  Additivity on the coordinate axes means the following: If $\ps{a}, \ps{b}, \ps{c} \in \ps{A}$ differ only on one factor $\ps{A}_i$, then the triangle inequality of the metric holds with equality. In mathematical terms:
  $$ \ps{a}_j = \ps{b}_j = \ps{c}_j \text{ for all } j \neq i \text{ and } (\ps{a}, \ps{c}) \succsim (\ps{a}, \ps{b}), (\ps{b}, \ps{c})  \Rightarrow \delta(\ps{a}, \ps{b}) + \delta(\ps{b}, \ps{c}) = \delta(\ps{a}, \ps{c}) $$
\end{rem}

In the following lemmas, until the proof of this theorem is finished, everything will be assumed as in \cref{thm-necc-suff-conds}, unless otherwise stated.

The following remark ties this result closer to the Minkowski metric, and provides a form which the metric must take. This will play an important role in the main result \cref{thm-uniquene-repr-minkowski}.

\begin{rem}[Form of $\delta$] \label{rem-metric-form}
	Examining \cref{proof-necc-suff-conds} more closely yields, that $G=F^{-1}$ must hold and the representation of $\delta$ takes the form
	\begin{equation*}
    \delta(\ps{a},\ps{b}) = F^{-1} \Big \{ \sum_{i=1}^{n} F(| \varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i) |) \Big \} \label{cor-delta-form}
	\end{equation*}
	where $F$ is continuous, convex and increasing.\\
	This follows form two arguments.
  Firstly, incorporating the $t_i$ into the $\varphi_i$ (which is possible because, by \cref{thm-repr-add-diff-struct}, the $\varphi_i$ are interval scales)
	\begin{align*}
    G \Big \{ \sum_{i=1}^{n} F_i(| \varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i) |) \Big \} & = G \Big \{ \sum_{i=1}^{n} F(t_i| \varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i) |) \Big \} \\
                                                                                        & = G \Big \{ \sum_{i=1}^{n} F(| \varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i) |) \Big \}
	\end{align*}
	yields the desired form inside the sum.

	Secondly, by using $F_i(x) = F_n(t_i x)$ and $F_i(x) = G^{-1}(t_i x)$.
	The first equation applied for $i=n$ yields $F_n(x) = F_n(t_n x)$ which implies $t_n = 1$. Using this in the second equation for $i=n$ yields $F(x) = F_n(x) = G^{-1}(t_n x) = G^{-1}(x)$. Hence, by taking the inverse on both sides (which is possible due to $F$ and $G$ both being increasing), we get the desired form for $\delta$.

\end{rem}


\begin{proof}[Proof of \crefpart{thm-necc-suff-conds}{thm-necc-cond}]\label{proof-necc-suff-conds}

  According to the assumptions of the theorem, the additive difference representaion $f(\ps{a},\ps{b})$ is compatible with a metric $\delta$ on $\ps{A}$, i.e.,
	$$
  f(\ps{a},\ps{b}) \leq f(\ps{c},\ps{d}) \iff \delta(\ps{a},\ps{b}) \leq \delta(\ps{c},\ps{d}) \quad \text{for all } \ps{a}, \ps{b}, \ps{c}, \ps{d} \in \ps{A} .
	$$
  Therefore, there exists an increasing function $G$ \footnotemark{} , such that for all $\ps{a}, \ps{b} \in \ps{A}$

	$$
  \delta(\ps{a}, \ps{b}) = G(f(\ps{a},\ps{b})) =  G \left( \sum_{i=1}^{n} F_i \left( |\varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i)| \right) \right).
	$$
  \footnotetext{A possible perspective on this is, that the metric $\metric$ induces an ordinal scale for pairs of elements. The same holds for the function $f$ and these orderings are the same. Because ordinal scales, by definition, are invariant under increasing bijections, there must exist an increasing function $G$, which transforms the $f$-scale into the $\metric$-scale.}
  Assume $\ps{a}$, $\ps{b}$, and $\ps{c}$ differ only in the $i$-th factor and $\ps{a}|\ps{b}|\ps{c}$. Thus, we can use additivity on the $i$-th coordinate axis for $\ps{a}, \ps{b}, \ps{c} \in \ps{A}$ which yields

	\[
    G \left( F_i \left( |\varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i)| \right) \right) + G \left( F_i \left( |\varphi_i(\ps{b}_i) - \varphi_i(\ps{c}_i)| \right) \right) = G \left( F_i \left( |\varphi_i(\ps{a}_i) - \varphi_i(\ps{c}_i)| \right) \right).
	\]

  Defining $|\varphi_i(\ps{a}_i) - \varphi_i(\ps{b}_i)| =: x$ and $|\varphi_i(\ps{b}_i) - \varphi_i(\ps{c}_i)| =: y$ yields

	\[
		G[F_i(x)] + G[F_i(y)] = G[F_i(x + y)].
	\]


  Further, define $H_i(x) = G[F_i(x)]$. This reduces the above equation to $H_i(x) + H_i(y) = H_i(x + y)$ for all $x, y$ in the domain of $F_i, 1 \leq i \leq n$. By \cref{thm-repr-add-diff-struct}, the domain is $\mathbb{R}$. This is Cauchy's functional equation and, under the assumption of monotonicity, the only solution is $H_i = t_i x$ for $t_i \in \mathbb{R}$, as shown in \parencite[p. 15]{aczelFunctionalEquationsSeveral1989}. Since the values are positive (as we are dealing with metrics), $H_i = t_i x$ with $t_i > 0$ must hold for all $i=1, ..., n$. Additionally, since $G$ is increasing, it has an inverse $G^{-1}$ satisfying

	\[
		G^{-1}[H_i(x)] = G^{-1}[t_i x] = F_i(x)
	\]

	where $F_i$ is defined. Since $G^{-1}$ is independent of $i$, the $F_i$ can only differ in range and in the unit of the domain (i.e. the scaling factor $t_i$ which defines the domain of $F_i$ by adjusting the domain of $G^{-1}$). Without loss of generality, we can assume that the range of $F_n$ is the largest among the $F_i$. This assumption allows us to define $F = F_n$ as the reference function, and the other $F_i$ can be expressed as $F_i(x) = F_n(t_i x) = F(t_i x)$ for $t_i > 0$ and for every $x$ in the domain of $F_i$.
	Continuity and convexity will be proven in the next lemmas.
\end{proof}

\begin{lem}\label{necc-lemma-ineq}
	$$F(\alpha + \beta + \gamma) + F(\gamma) \geq F(\alpha + \gamma) + F(\beta + \gamma).$$
\end{lem}

\begin{proof}

	In the triangle inequality,

	\[
		F^{-1} \left[ \sum_{i=1}^{n} F(\alpha_i) \right] + F^{-1} \left[ \sum_{i=1}^{n} F(\beta_i) \right] \geq F^{-1} \left[ \sum_{i=1}^{n} F(\alpha_i + \beta_i) \right],
	\]

	let $\alpha_1 = \gamma, \alpha_2 = F^{-1}(F(\alpha + \gamma) - F(\gamma)), \alpha_j = 0$ for $3 \leq j \leq n$ and $\beta_1 = \beta, \beta_j = 0$ for $2 \leq j \leq n$. Hence,

	\begin{align*}
		F(\alpha + \beta + \gamma) & = F \left[ F^{-1} \left[ \sum_{i=1}^{n} F(\alpha_i) \right] + F^{-1} \left[ \sum_{i=1}^{n} F(\beta_i) \right] \right] \\
		                           & \geq \sum_{i=1}^{n} F(\alpha_i + \beta_i) \tag{by the triangle inequality}                                            \\
		                           & = F(\beta + \gamma) + F(\alpha + \gamma) - F(\gamma).
	\end{align*}
\end{proof}

\begin{lem}\label{lemma-continuity}
	$F$ is continuous.
\end{lem}

\begin{proof}


  Since $F$ is increasing and defined on an interval, all discontinuities are gaps \parencite[p. 335]{elementaryAna} \footnotemark. Real analysis yields, that real monotonic functions have at most countable many gaps. We will derive a contradiction from this.

  \footnotetext{\parencite{elementaryAna} prove the following statement: Let $f$ be a real-valued monotonic function defined on an interval $[a, b]$. Then the set of points of discontinuity of $f$ in that interval is countable.\\ Combining this statement with the paragraph above it, yields that these discontinuities must be gaps.\\
  Since any interval (open, half-open and unbouded intervals) can be represented by a countable union of closed intervals, this theorem can be extended to arbitrary intervals since countable unions of countable sets are still countable.\\
This can be done by using $(a,b) = \bigcup_{n=1}^{\infty}[a+1/n, b-1/n]$, or $(a, \infty) = \bigcup_{n=1}^{\infty}[a+1/n, a+n]$. Similar arguments apply to other types of intervals.}

	Assume $F(\alpha)$ is the lower boundary of a gap, then there exists $\epsilon > 0$ such that for all $\delta > 0$, $F(\alpha + \delta) - F(\alpha) > \epsilon$. However, by \cref{necc-lemma-ineq}, for every $\beta > 0$,
	\[
		F(\alpha + \beta + \delta) - F(\alpha + \beta) \geq F(\alpha + \delta) - F(\alpha) > \epsilon,
	\]

	and so there is a gap at any $\alpha + \beta$, and $F$ would have to be discontinuous everywhere. This is impossible, so $F$ must be continuous.
\end{proof}

\begin{lem}[Characterization of Convexity]
	Suppose $F$ is a strictly increasing function defined on the nonnegative reals. Then $F$ is convex iff for all $\alpha, \beta, \gamma \geq 0$ the inequality from \ref{necc-lemma-ineq} holds, i.e.,
	$$
		F(\alpha + \beta + \gamma) + F(\gamma) \geq F(\alpha + \gamma) + F(\beta + \gamma).
	$$
	\begin{proof}

		\begin{proofpart}[$\Rightarrow$]

			Suppose, $F$ is convex, and let $p = \alpha / (\alpha + \beta)$, then

			\begin{align*}
				p(\alpha + \beta + \gamma) + (1 - p)\gamma & = \alpha + \gamma \\
				(1 - p)(\alpha + \beta + \gamma) + p\gamma & = \beta + \gamma.
			\end{align*}

			Hence, by convexity,

			\begin{align*}
				F(\alpha + \gamma) + F(\beta + \gamma) & = F \left[ p (\alpha + \beta + \gamma) + (1 - p) \gamma \right] + F \left[ (1 - p) (\alpha + \beta + \gamma) + p \gamma \right] \\
				                                       & \leq p F(\alpha + \beta + \gamma) + (1 - p) F(\gamma) + (1 - p) F(\alpha + \beta + \gamma) + p F(\gamma)                        \\
				                                       & = F(\alpha + \beta + \gamma) + F(\gamma).
			\end{align*}
		\end{proofpart}

		\begin{proofpart}[$\Leftarrow$]
			To prove the converse, we will use the continuity of $F$ by \cref{lemma-continuity}. Suppose, $\alpha > \beta$, then

			\begin{align*}
				F(\alpha) + F(\beta) & = F \left( \frac{\alpha - \beta}{2} + \frac{\alpha - \beta}{2} + \beta \right) + F(\beta) \\
				                     & \geq 2 F \left( \frac{\alpha - \beta}{2} + \beta \right) \tag{by \cref{lemma-continuity}} \\
				                     & = 2 F \left( \frac{\alpha + \beta}{2} \right),
			\end{align*}

      which implies convexity for a continuous $F$ \parencite[p. 51, Exercise 6]{lecturesConvGeo}.
		\end{proofpart}
	\end{proof}
\end{lem}
