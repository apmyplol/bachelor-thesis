\begin{ex}[p-exp Metric Family]\label{def-p-exp-metric-family} \index{p-exp metric family}
	Let the $F$ in the AD model (\cref{def-add-diff-model}) be given by
	\begin{equation}\label{def-p-exp-eq}
		F(x) = \q \cdot (e^{\p \cdot x} - 1)^\r \qquad \q, \p > 0, \r \geq 1
	\end{equation}
  and let $G$ be its inverse\footnote{Since we prove that this functions satisfies the conditions of Mulholland's inequality, the function is increasing and therefore is invertible.}.
	The model defined by this $F$ is a metric.
\end{ex}

\begin{proof}
	First, we will prove positivity and symmetry of this function, and then turn to the triangle inequality which is a result of Mulholland's theorem.
	\begin{proofpart}[Positivity and Symmetry]
		Positivity and symmetry are both straightforward.
		\begin{enumerate}
			\item Positivity: $\delta(\ve{x},\ve{y}) = 0$ implies $F(|x_i - y_i|) = 0$ for all $i=1, \dots, n$, which is only possible if $|x_i - y_i| \cdot r = 0$, meaning $x_i = y_i$ for all $i=1, \dots, n$.\\
			      Conversely, if $\ve{x} \neq \ve{y}$, then $x_i \neq y_i$ for at least one $i$ in $1, \dots n$, which implies $F_i(|x_i - y_i|) > 0$, and thus $\delta(\ve{x},\ve{y}) > 0$.
			\item Symmetry: Symmetry follows directly from the fact that $|x_i - y_i| = |y_i - x_i|$ holds.
		\end{enumerate}
	\end{proofpart}
	The triangle inequality follows from Mulholland's Inequality \parencite[p. 297]{mulholland}, which we will restate here for completeness:

	\begin{thm}[Mulholland's Inequality]
		The inequality
		$$
			f^{-1} \left[ \sum_{i=1}^n f(|x_i + y_i|) \right] \leq f^{-1}  \left[ \sum_{i=1}^n f(|x_i|) \right] + f^{-1} \left[ \sum_{i=1}^n f(|y_i|) \right]
		$$
		holds for all $\ve{x}, \ve{y} \in \mathbb{R}^n, n = 1, 2, \ldots$ whenever the following three conditions are satisfied:
		\begin{enumerate}
			\item $f$ is continuous, increasing and $f(0) = 0$.
			\item $f$ is convex on $[0, \infty)$.
			\item $\log f(e^x)$ is convex for $x \in \mathbb{R}$.
		\end{enumerate}
	\end{thm}

	% \setcounter{proofpart}{1}
	\begin{proofpart}[Proof that \cref{def-p-exp-eq} satisfies Mulholland's conditions]
		Regarding point 1: Continuity and $F(0) = 0$ are clear. For monotonicity, note that $e^{\p x} - 1$ is increasing, exponentiation is also increasing, and multiplying by $\q > 0$ preserves monotonicity. Thus, the composition is also increasing.\\
		Regarding point 2: Similar reasoning applies for convexity. The composition of convex functions is convex, if the outer function is non-decreasing \parencite[p. 481, Note 207]{elementaryAna}. Since multiplication by a positive constant and exponentiation for exponents greater than $1$ is both convex and increasing, the resulting function must be convex.

		We now proceed to prove the convexity of $\log F(e^x)$.
		By logarithmic properties, it holds that
		$$ \log \q(e^{\p e^x} -1)^\r = \log \q + \r \cdot  \log(e^{\p e^x} - 1). $$
		Arbitrary addition and multiplication by $\r \geq 1$ preserve convexity, so it suffices to show the convexity of $\log(e^{\p e^x} - 1)$.
		We prove this by examining the second derivative. If the second derivative is strictly positive, then the function is convex \parencite[p. 57, Remark 2.13]{lecturesConvGeo}.
		Differentiating twice yields
		$$
			\frac{\p  e^{\p e^x + x} \left( \p  (-e^x) + e^{\p e^x} - 1 \right)}{\left( e^{\p e^x} - 1 \right)^2} \overset{!}{>} 0.
		$$
		The denominator is always positive and non-zero, so we can ignore it. Similarly, the factor $\p  e^{\p e^x + x}$ is always positive and can also be omitted. It remains to show that
			$$-\p  e^x + e^{\p e^x} - 1 > 0,$$
    or equivalently, 
			$$e^{\p e^x}  - \p  e^x     > 1$$
		holds.
		This can be shown in two steps. First, we show that the left-hand side is strictly increasing. Second, we prove that its limit as $x$ approaches $-\infty$ is $1$.
		Therefore, the inequality holds for all $x \in \mathbb{R}$.

		\begin{enumerate}
			\item For the first point, we show that the derivative is greater than zero. Differentiation yields
			      $$ \p e^x \cdot e^{\p e^x} - \p e^x = \p e^x \cdot (e^{\p e^x} - 1) > 0. $$ The inequality holds, since $e^x > 0$ for all $x \in \mathbb{R}$ and $e^x > 1$ for $x > 0$.
			\item For the second point, we calculate the limit as $x$ tends to $-\infty$:
			      $$ \lim_{x \rightarrow -\infty} e^{\p e^x}  - \p  e^x = \lim_{x \rightarrow -\infty} e^{\p e^x}  - \lim_{x \rightarrow -\infty} \p  e^x =  1 - 0 = 1. $$

			      Thus, convexity is established, and the claim follows from Mulholland's Inequality.
		\end{enumerate}
	\end{proofpart}
\end{proof}

\begin{rem}[Minkowski metric is part of the p-exp metric family] \label{rem-mink-pexp}
	If we set $\q = \p ^{-\r}$ in (\cref{def-p-exp-eq}), we obtain the Minkowski metric in the limit as $\p  \rightarrow 0$.
\end{rem}

\begin{proof}
	Setting $\q = \p ^{-\r}$ yields
	$$ \p ^{-\r}(e^{\p x} - 1)^\r = \left( \frac{e^{\p x} -1}{\p } \right)^\r .$$
	As $\p  \rightarrow 0$, both the numerator and denominator tend to $0$, therefore we can apply L'Hospital's rule.
	Hence,
	\begin{align*}
		\lim_{\p  \rightarrow 0} \left( \frac{e^{\p x} -1}{\p } \right)^\r & = \left( \lim_{\p  \rightarrow 0} \frac{e^{\p x} -1}{\p } \right)^\r \tag{continuity}              \\
		                                                                   & = \left( \lim_{\p  \rightarrow 0} \frac{x \cdot e^{\p x}}{1} \right)^\r \tag{L'Hôpital's rule} \\
		                                                                   & = x^\r.
	\end{align*}
\end{proof}


Not every function that can be represented by the additive difference model is a metric, see \cref{ex-not-metric}.
