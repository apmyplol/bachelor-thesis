\section{Examples}

\begin{ex}[p-exp Metric Family]\label{def-p-exp-metric-family}
	Let the $F$ in the additive difference model (\ref{def-add-diff-model}) be given by
	\begin{equation}\label{def-p-exp-eq}
		F(t) = q \cdot (e^{t \cdot r} - 1)^p \qquad q, r > 0, p \geq 1.
	\end{equation}
	The model defined by this $F$ is a metric.
\end{ex}


\begin{proof}
	First, we will prove positivity and symmetry of this function, and then turn to the triangle inequality which is a result of theorem which provides a generalization of the Mikowski inequality.
	\begin{proofpart}[Positivity and Symmetry]
		Positivity and symmetry are both straight forward.
		\begin{enumerate}
			\item Positivity: $\delta(a,b) = 0$ implies $F(|a_i - b_i|) = 0$ for all $i=1, \dots, n$, which is only possible if $|a_i - b_i| \cdot r = 0$, meaning $a_i = b_i$ for all $i=1, \dots, n$.\\
			      Conversely, if $a \neq b$, then $a_i \neq b_i$ for at least one $i$ in $1, \dots n$, which implies $F_i(|a_i - b_i|) > 0$, and thus $\delta(a,b) > 0$.
			\item Symmetry: Symmetry follows directly from the fact that $|a_i - b_i| = |b_i - a_i|$ holds.
		\end{enumerate}
	\end{proofpart}

	The triangle inequality follows from Mullholland's theorem \parencite[p.~297]{mulholland}.

	\begin{thm}[Mulholland's Inequality]
		The inequality
		$$
			f^{-1} \left[ \sum_{i=1}^n f(|x_i + y_i|) \right] \leq f^{-1}  \left[ \sum_{i=1}^n f(|x_i|) \right] + f^{-1} \left[ \sum_{i=1}^n f(|y_i|) \right].
		$$
		holds for all $x, y \in \mathbb{R}^n, n = 1, 2, \ldots$ whenever the following three conditions are satisfied:
		\begin{enumerate}
			\item $f$ is continuous, increasing and $f(0) = 0$.
			\item $f$ is convex on $[0, \infty)$.
			\item $\log f(e^x)$ is convex for $x \in (-\infty, \infty)$.
		\end{enumerate}
	\end{thm}

	% \setcounter{proofpart}{1}
	\begin{proofpart}[Proof that \cref{def-p-exp-eq} satisfies Mulholland's conditions]

		Regarding Point 1: Continuity and $f(0) = 0$ are clear. For monotonicity, note that $e^{rt} - 1$ is increasing, exponentiation is also increasing, and multiplying by $q > 0$ preserves monotonicity. Thus, the composition is also increasing.\\
    Regarding point 2: Similar reasoning applies for convexity. The composition of convex functions is convex, if the outer function is non-decreasing \parencite[p. 481, Note 207]{elementaryAna}. Since multiplication by a positive constant and exponentiation for exponents greater than $1$ is both convex and increasing, the resulting function must be convex. \\

		We now proceed to prove the convexity of $\log F(e^t)$.
		By logarithmic properties, it holds that
		$$ \log q(e^{re^x} -1)^p = \log q + p \cdot  \log(e^{re^x} - 1). $$
		Arbitrary addition and multiplication by $p \geq 1$ preserve convexity, so it suffices to show the convexity of $\log(e^{re^x} - 1)$.

    We prove this by examining the second derivative. If the second derivative is strictly positive, the function is convex \parencite[p. 57, Remark 2.13]{lecturesConvGeo}.

		Differentiating twice yields
		$$
			\frac{r e^{re^x + x} \left( r (-e^x) + e^{re^x} - 1 \right)}{\left( e^{re^x} - 1 \right)^2} \overset{!}{>} 0.
		$$
		The denominator is always positive and non-zero, so we can ignore it. Similarly, the factor $r e^{re^x + x}$ is always positive and can also be omitted. It remains to show that
		\begin{align*}
			-r e^x + e^{re^x} - 1   & > 0 \\
			\iff  e^{re^x}  - r e^x & > 1
		\end{align*}
		holds.

		This can be shown in two steps. First, we show that the left-hand side is strictly increasing. Second, we prove that its limit as $x \rightarrow -\infty$ is $1$.
		Therefore, the inequality holds for all $x \in \mathbb{R}$.

		\begin{enumerate}
			\item For the first point, we show that the derivative is greater than zero. Differentiation yields
			      $$ re^x \cdot e^{re^x} - re^x = re^x \cdot (e^{re^x} - 1) > 0 $$ since $e^x > 0$.
			\item
			      $$ \lim_{x \rightarrow -\infty} e^{re^x}  - r e^x = \lim_{x \rightarrow -\infty} e^{re^x}  - \lim_{x \rightarrow -\infty} r e^x =  1 - 0 = 1 $$

			      Thus, convexity is established, and the claim follows from Mulholland's theorem.
		\end{enumerate}

	\end{proofpart}

\end{proof}

\begin{rem}[Minkowski metric is part of the p-exp metric family]

	If we set $q = r^{-p}$ in (\ref{def-p-exp-eq}), we obtain the Minkowski metric in the limit as $r \rightarrow 0$.

	\begin{proof}
		Setting $q = r^{-p}$ yields
		$$ r^{-p}(e^{rt} - 1)^p = \left( \frac{e^{rt} -1}{r} \right)^p .$$
		As $r \rightarrow 0$, both the numerator and denominator tend to $0$ (we can pull the limit inside the exponent since the exponentiation is continuous), therefore we can apply L'Hôpital's rule.
		Hence
		\begin{align*}
			\lim_{r \rightarrow 0} \left( \frac{e^{rt} -1}{r} \right)^p & = \left( \lim_{r \rightarrow 0} \frac{e^{rt} -1}{r} \right)^p \tag{continuity}                \\
			                                                            & = \left( \lim_{r \rightarrow 0} \frac{t \cdot e^{rt} - 1}{1} \right)^p \tag{L'Hôpital's rule} \\
			                                                            & = t^p.
		\end{align*}
	\end{proof}

\end{rem}
